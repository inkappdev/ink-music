# ink music's exact ways: its own to a USB DAC (usbaudio.c: plain C on Linux's usbfs) and AAudio exclusive for a player's own DAC (exclusive.c)
cmake_minimum_required(VERSION 3.22)
project(inkusb C)
add_library(inkusb SHARED usbaudio.c exclusive.c)
target_compile_options(inkusb PRIVATE -O2 -Wall)
target_link_libraries(inkusb log m aaudio)
